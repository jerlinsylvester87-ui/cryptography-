#include <stdio.h>
#include <string.h>
#include <openssl/evp.h>

int main()
{
    unsigned char key[24] = "123456789012345678901234";
    unsigned char iv[8] = "12345678";
    unsigned char plaintext[] = "Hello CBC Mode";
    unsigned char ciphertext[100];
    int len, ciphertext_len;

    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();

    EVP_EncryptInit_ex(ctx, EVP_des_ede3_cbc(), NULL, key, iv);

    EVP_EncryptUpdate(ctx, ciphertext, &len,
                      plaintext, strlen((char *)plaintext));
    ciphertext_len = len;

    EVP_EncryptFinal_ex(ctx, ciphertext + len, &len);
    ciphertext_len += len;

    printf("Ciphertext: ");
    for(int i = 0; i < ciphertext_len; i++)
        printf("%02X", ciphertext[i]);

    printf("\n");

    EVP_CIPHER_CTX_free(ctx);
    return 0;
}

Input:
Plaintext: Hello CBC Mode
Key: 123456789012345678901234
IV: 12345678

Output:
Ciphertext: 88D88E49551ECF8162D185B9474B3A99
