
#include <stdio.h>
#include <string.h>

int main()
{
    char p[] = "SENDMOREMONEY";
    char target[] = "CASHNOTNEEDED";
    int key[13] = {9,0,1,7,23,15,21,14,11,11,2,8,9};
    int i, c[13], k2[13];

    printf("Encrypted text: ");

    for(i = 0; i < 13; i++)
    {
        c[i] = (p[i] - 'A' + key[i]) % 26;
        printf("%c", c[i] + 'A');
    }

    printf("\nKey for target plaintext: ");

    for(i = 0; i < 13; i++)
    {
        k2[i] = (c[i] - (target[i] - 'A') + 26) % 26;
        printf("%d ", k2[i]);
    }

    printf("\n");
    return 0;
}


Input:
Plaintext: SEND MORE MONEY
Key stream: 9 0 1 7 23 15 21 14 11 11 2 8 9
Target plaintext: CASH NOT NEEDED

Output:
Encrypted text: BEOKJDMSXZPMH
Key for target plaintext: 25 4 22 3 22 15 19 5 19 21 12 8 4
