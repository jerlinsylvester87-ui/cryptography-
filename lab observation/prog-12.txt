
#include <stdio.h>
#include <string.h>

int main()
{
    int inv[2][2] = {{5,12},{15,25}};
    char text[] = "UKIXUKYDROMEIWSZXWIOKUNUKHXHROAJROANQYEBTLKJEGAD";
    int i, a, b;

    printf("Decrypted Message: ");

    for(i = 0; text[i] != '\0'; i += 2)
    {
        a = text[i] - 'A';
        b = text[i+1] - 'A';

        printf("%c%c",
        (inv[0][0]*a + inv[0][1]*b) % 26 + 'A',
        (inv[1][0]*a + inv[1][1]*b) % 26 + 'A');
    }

    printf("\n");
    return 0;
}


Input:
UKIXUKYDROMEIWSZXWIOKUNUKHXHROAJROANQYEBTLKJEGAD

Output:
MEETMEATTHEUSUALPLACEATTENRATHERTHANEIGHTOCLOCKX
