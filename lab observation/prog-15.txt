
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main()
{
    char text[500];
    int shift, i, n;

    printf("Enter ciphertext: ");
    fgets(text, sizeof(text), stdin);

    printf("Enter number of possible plaintexts: ");
    scanf("%d", &n);

    printf("\nPossible plaintexts:\n");

    for(shift = 0; shift < n && shift < 26; shift++)
    {
        printf("Key %2d: ", shift);

        for(i = 0; text[i] != '\0'; i++)
        {
            if(isalpha(text[i]))
            {
                char ch = toupper(text[i]);
                printf("%c", (ch - 'A' - shift + 26) % 26 + 'A');
            }
            else
                printf("%c", text[i]);
        }

        printf("\n");
    }

    return 0;
}

Enter ciphertext: KHOOR ZRUOG
Enter number of possible plaintexts: 10
